# provision_vps.sh — HermesSell VPS bootstrap
# Ejecutar: bash provision_vps.sh
set -euo pipefail
# 1. Sistema base
apt-get update && apt-get upgrade -y
apt-get install -y \
  curl git wget unzip build-essential \
  python3.11 python3.11-venv python3.11-dev python3-pip \
  ffmpeg ripgrep fd-find jq \
  postgresql-16 postgresql-client-16 \
  redis-server nginx certbot python3-certbot-nginx \
  ufw fail2ban htop tmux
# 2. Docker
curl -fsSL https://get.docker.com | bash
apt-get install -y docker-compose-plugin
# 3. Usuario hermesell (nunca correr como root)
useradd -m -s /bin/bash -G docker,sudo hermesell
echo 'hermesell ALL=(ALL) NOPASSWD:/usr/bin/docker' >> /etc/sudoers.d/hermesell
# 4. Node.js 20 LTS (para Kapso gateway)
curl -fsSL https://deb.nodesource.com/setup_20.x | bash -
apt-get install -y nodejs
# 5. Firewall
ufw default deny incoming
ufw default allow outgoing
ufw allow ssh
ufw allow 80/tcp
ufw allow 443/tcp
ufw --force enable
# 6. PostgreSQL — base de datos del SaaS
sudo -u postgres psql <<SQL
CREATE USER hermesell WITH PASSWORD 'CAMBIAR_ESTE_PASSWORD';
CREATE DATABASE hermesell_saas OWNER hermesell;
CREATE DATABASE hermesell_hindsight OWNER hermesell;
SQL
# 7. Redis para Celery
systemctl enable --now redis-server
# 8. Directorios de datos por tenant
mkdir -p /opt/hermesell/tenants
mkdir -p /opt/hermesell/uploads
mkdir -p /opt/hermesell/logs
chown -R hermesell:hermesell /opt/hermesell
echo '=== Provisioning completo. Cambiá el password de PostgreSQL. ==='

Cambiá el password de PostgreSQL antes de continuar.
El usuario hermesell nunca tiene acceso directo a los datos de otros tenants.
Cada tenant vive en /opt/hermesell/tenants/{client_id}/ con permisos 700.

Webhook handler en FastAPI (services/api/routers/webhook.py)
Este es el código real del webhook que recibe eventos de Meta y los despacha al agente correcto:
from fastapi import APIRouter, Request, HTTPException, Header
from hermesell.whatsapp.webhook import WebhookHandler
from hermesell.tenant import TenantManager
import hmac, hashlib, os
router = APIRouter(prefix='/webhook', tags=['webhook'])
handler = WebhookHandler()
manager = TenantManager()
@router.get('/')
async def verify(hub_mode: str, hub_challenge: str, hub_verify_token: str):
    if hub_verify_token != os.getenv('VERIFY_TOKEN'):
        raise HTTPException(403)
    return int(hub_challenge)
@router.post('/')
async def receive(
    request: Request,
    x_hub_signature_256: str = Header(None)
):
    body = await request.body()
    # verifySignature — igual que @kapso/whatsapp-cloud-api/server
    secret = os.getenv('META_APP_SECRET', '').encode()
    expected = 'sha256=' + hmac.new(secret, body, hashlib.sha256).hexdigest()
    if not hmac.compare_digest(expected, x_hub_signature_256 or ''):
        raise HTTPException(401, 'Invalid signature')
    payload = await request.json()
    events = handler.normalize(payload)
    for msg in events.get('messages', []):
        phone_id = msg['metadata']['phone_number_id']
        tenant = await manager.get_by_phone_id(phone_id)
        if tenant:
            await manager.dispatch(tenant, msg)
    return {'status': 'ok'}
hermesell/whatsapp/webhook.py — normalizeWebhook en Python
class WebhookHandler:
    def normalize(self, payload: dict) -> dict:
        """Replica la lógica de normalizeWebhook de @kapso/whatsapp-cloud-api"""
        messages, statuses, calls = [], [], []
        for entry in payload.get('entry', []):
            for change in entry.get('changes', []):
                value = change.get('value', {})
                meta = {
                    'phone_number_id': value.get('metadata', {}).get('phone_number_id'),
                    'display_phone_number': value.get('metadata', {}).get('display_phone_number'),
                }
                for m in value.get('messages', []):
                    m['metadata'] = meta
                    m['kapso'] = {'direction': 'inbound'}
                    messages.append(m)
                statuses.extend(value.get('statuses', []))
        return {'messages': messages, 'statuses': statuses, 'calls': calls}

services/preprocessor/tasks/video_task.py — extracción multimodal
Este es el task más complejo. Extrae frames clave del video, los envía a Gemini Vision y combina el transcript de audio con el análisis visual para producir hechos estructurados:
from celery import shared_task
from hermesell.ingestion.extractors.video import VideoExtractor
from hermesell.ingestion.hindsight import HindsightIngestor
@shared_task(bind=True, max_retries=3, default_retry_delay=60)
def process_video(self, file_path: str, tenant_id: str, metadata: dict):
    try:
        extractor = VideoExtractor()
        # 1. Extraer frames clave (1 por cada 30s)
        frames = extractor.extract_keyframes(file_path, interval_sec=30)
        # 2. Transcript del audio con Whisper
        transcript = extractor.transcribe_audio(file_path)
        # 3. Análisis visual de cada frame con Gemini Flash
        visual_facts = []
        for frame in frames:
            facts = extractor.analyze_frame_vision(
                frame,
                prompt=(
                    'Sos un extractor de datos para un catálogo de ventas. '
                    'Describí todos los productos visibles: nombre, color, '
                    'características, precio si aparece, texto superpuesto. '
                    'Responde SOLO en JSON: {"productos": [...], "textos": [...]}'
                )
            )
            visual_facts.extend(facts)
        # 4. Combinar transcript + hechos visuales
        combined = extractor.merge_facts(transcript, visual_facts)
        # 5. Ingestar en Hindsight del tenant
        ingestor = HindsightIngestor(tenant_id=tenant_id)
        ingestor.ingest_facts(combined, source=file_path, metadata=metadata)
        return {'status': 'ok', 'facts_ingested': len(combined)}
    except Exception as exc:
        raise self.retry(exc=exc)
hermesell/ingestion/extractors/video.py
import whisper
from moviepy.editor import VideoFileClip
from openai import OpenAI
import base64, json, tempfile
from pathlib import Path
class VideoExtractor:
    def __init__(self):
        self.whisper_model = whisper.load_model('base')
        self.vision_client = OpenAI(
            base_url='https://openrouter.ai/api/v1',
            api_key=os.getenv('OPENROUTER_API_KEY')
        )
    def extract_keyframes(self, path: str, interval_sec=30) -> list[str]:
        clip = VideoFileClip(path)
        frames = []
        for t in range(0, int(clip.duration), interval_sec):
            with tempfile.NamedTemporaryFile(suffix='.jpg', delete=False) as f:
                clip.save_frame(f.name, t=t)
                frames.append(f.name)
        clip.close()
        return frames
    def transcribe_audio(self, path: str) -> str:
        result = self.whisper_model.transcribe(path)
        return result['text']
    def analyze_frame_vision(self, frame_path: str, prompt: str) -> list[dict]:
        with open(frame_path, 'rb') as f:
            b64 = base64.b64encode(f.read()).decode()
        resp = self.vision_client.chat.completions.create(
            model='google/gemini-2.5-flash',
            messages=[{
                'role': 'user',
                'content': [
                    {'type': 'image_url', 'image_url': {'url': f'data:image/jpeg;base64,{b64}'}},
                    {'type': 'text', 'text': prompt}
                ]
            }],
            response_format={'type': 'json_object'}
        )
        raw = json.loads(resp.choices[0].message.content)
        return raw.get('productos', []) + raw.get('textos', [])
hermesell/ingestion/hindsight.py — HindsightIngestor
import psycopg2, json, os
from datetime import datetime
class HindsightIngestor:
    """Escribe hechos estructurados en el grafo de conocimiento de Hindsight."""
    def __init__(self, tenant_id: str):
        self.tenant_id = tenant_id
        self.conn = psycopg2.connect(os.getenv('HINDSIGHT_DB_URL'))
    def ingest_facts(self, facts: list[dict], source: str, metadata: dict = None):
        """Cada fact es un dict con {entity, attribute, value}."""
        with self.conn.cursor() as cur:
            for fact in facts:
                cur.execute(
                    '''INSERT INTO facts
                    (tenant_id, entity, attribute, value, source, ingested_at, meta)
                    VALUES (%s, %s, %s, %s, %s, %s, %s)''',
                    (
                        self.tenant_id,
                        fact.get('entity', 'unknown'),
                        fact.get('attribute', 'unknown'),
                        str(fact.get('value', '')),
                        source,
                        datetime.utcnow(),
                        json.dumps(metadata or {})
                    )
                )
        self.conn.commit()

# Sales Closer — Protocolo de cierre de ventas
## Cuándo activar
- El cliente pregunta precio, stock, características de un producto
- El cliente dice frases como 'lo quiero', 'me interesa', 'cómo compro'
- El objetivo actual (/goal) es cerrar una venta
## Flujo obligatorio (en orden)
### Paso 1 — Identificar el producto
Usar hindsight_recall con la entidad mencionada por el cliente.
Si hay ambigüedad, preguntar una vez, solo una.
### Paso 2 — Verificar stock
hindsight_recall(query='stock de {producto}')
Si stock = 0, ofrecer alternativa similar o notificar al home channel.
NUNCA inventar stock disponible.
### Paso 3 — Presentar precio
hindsight_recall(query='precio de {producto}')
Formato: '[Producto] — ${precio}. [Beneficio clave en 1 línea].'
Máximo 2 beneficios. Sin texto de relleno.
### Paso 4 — Manejar objeciones
Si el cliente dice 'es caro': mostrar comparativa con alternativa más barata.
Si el cliente dice 'lo pienso': preguntar qué le genera dudas (solo una vez).
NUNCA ofrecer descuento sin autorización del home channel.
### Paso 5 — Confirmar método de pago
Ofrecer los métodos en este orden: transferencia, MercadoPago, efectivo.
Confirmar datos de pago del catálogo via hindsight_recall.
### Paso 6 — Registrar venta
hindsight_retain(entity='{producto}', attribute='stock',
  value=str(stock_anterior - cantidad))
Enviar confirmación al cliente y notificación al home channel.
## Reglas absolutas
- Respuestas de máximo 3 líneas en el primer contacto
- Nunca inventar precios, stock ni características
- Si hindsight_recall no devuelve datos: 'Verifico y te confirmo en un momento'
- Si el agente se bloquea 2 veces: kanban_block() y notificar al humano

# Lead Qualifier
## Señales de clasificación
HOT: 'quiero comprar', 'cuánto cuesta', 'tienen stock', 'cómo pago'
WARM: 'me puede contar de', 'qué diferencia hay', 'tienen X producto'
COLD: 'solo mirando', 'es para más adelante', 'me mandás el catálogo'
SUPPORT: 'mi pedido', 'no llegó', 'quiero devolver', 'tengo un problema'
## Protocolo
1. Leer el primer mensaje del cliente
2. Clasificar con honcho_profile si hay historial
3. Responder con saludo personalizado + pregunta de calificación
4. Despachar al skill correcto via Kanban:
   - HOT/WARM → sales-closer
   - COLD → catalog-lookup
   - SUPPORT → kanban_block() + notificar humano

El cliente principal del SDK. Punto de entrada para implementadores:
from __future__ import annotations
import os
from pathlib import Path
from .tenant import TenantManager
from .ingestion.preprocessor import Preprocessor
from .agent.soul import SOULBuilder
from .agent.hermes import HermesAgentWrapper
from .whatsapp.gateway import KapsoGateway
class HermesSellClient:
    """
    Cliente principal de HermesSell.
    Uso básico:
        from hermesell import HermesSellClient
        client = HermesSellClient.from_env()
        tenant = client.create_tenant(
            name='Mi Tienda',
            phone_number_id='123456',
            access_token='EAAx...',
        )
        client.ingest(tenant.id, 'catalogo.pdf')
    """
    def __init__(
        self,
        openrouter_api_key: str,
        kapso_api_key: str,
        db_url: str,
        redis_url: str = 'redis://localhost:6379/0',
        tenants_path: str = '/opt/hermesell/tenants',
    ):
        self.openrouter_api_key = openrouter_api_key
        self.kapso_api_key = kapso_api_key
        self.db_url = db_url
        self.redis_url = redis_url
        self.tenants_path = Path(tenants_path)
        self._tenants = TenantManager(db_url, tenants_path)
        self._preprocessor = Preprocessor(redis_url)
        self._gateway = KapsoGateway(kapso_api_key)
    @classmethod
    def from_env(cls) -> HermesSellClient:
        """Construye el cliente desde variables de entorno."""
        return cls(
            openrouter_api_key=os.environ['OPENROUTER_API_KEY'],
            kapso_api_key=os.environ['KAPSO_API_KEY'],
            db_url=os.environ['DATABASE_URL'],
            redis_url=os.getenv('REDIS_URL', 'redis://localhost:6379/0'),
            tenants_path=os.getenv('TENANTS_PATH', '/opt/hermesell/tenants'),
        )
    def create_tenant(
        self,
        name: str,
        phone_number_id: str,
        access_token: str,
        waba_id: str,
        soul_config: dict | None = None,
    ) -> 'Tenant':
        """Crea un nuevo tenant: DB + Docker + Hindsight + SOUL.md."""
        return self._tenants.create(
            name=name,
            phone_number_id=phone_number_id,
            access_token=access_token,
            waba_id=waba_id,
            soul_config=soul_config or {},
            openrouter_key=self.openrouter_api_key,
        )
    def ingest(self, tenant_id: str, file_path: str) -> str:
        """Encola un archivo para preprocesamiento asíncrono. Devuelve task_id."""
        return self._preprocessor.enqueue(tenant_id, file_path)
    def ingest_status(self, task_id: str) -> dict:
        """Estado de una tarea de preprocesamiento."""
        return self._preprocessor.status(task_id)
    def send_message(self, tenant_id: str, to: str, text: str) -> dict:
        """Envía un mensaje WhatsApp desde el número del tenant."""
        tenant = self._tenants.get(tenant_id)
        return self._gateway.send_text(
            phone_number_id=tenant.phone_number_id,
            access_token=tenant.access_token,
            to=to,
            body=text,
        )
    def list_tenants(self) -> list:
        return self._tenants.list_all()
    def delete_tenant(self, tenant_id: str, destroy_data: bool = False):
        """Elimina el tenant. destroy_data=True borra el volumen Docker."""
        self._tenants.delete(tenant_id, destroy_data)

Flujo completo de onboarding de un nuevo cliente
El cliente entra al dashboard, hace clic en 'Conectar WhatsApp' y en menos de 5 minutos su agente está activo. No toca ninguna terminal.
Cliente se registra en /signup con email, nombre empresa y plan
El dashboard muestra el botón 'Conectar WhatsApp'
Al hacer clic, se abre el popup Meta Embedded Signup (Facebook SDK)
El cliente loguea con Facebook → selecciona Business Portfolio → elige número
Meta llama al callback de tu app con access_token + waba_id + phone_number_id
El backend llama a HermesSellClient.create_tenant() con esos datos
create_tenant() hace: (a) registra en DB, (b) crea directorio Docker, (c) genera SOUL.md, (d) inicializa Hindsight, (e) levanta el contenedor
El cliente es redirigido a su dashboard con estado 'Agente activo'
El cliente sube su catálogo (CSV/PDF/video) y el worker lo procesa en background
En cuanto Hindsight tiene al menos 1 hecho, el agente empieza a responder

from pydantic import BaseModel
from hermesell import HermesSellClient
from cryptography.fernet import Fernet
import os
router = APIRouter(prefix='/tenants', tags=['tenants'])
fernet = Fernet(os.environ['ENCRYPTION_KEY'].encode())
class MetaCallbackPayload(BaseModel):
    access_token: str
    phone_number_id: str
    waba_id: str
    user_id: str         # ID del usuario en tu sistema
    business_name: str
@router.post('/connect-whatsapp')
async def connect_whatsapp(payload: MetaCallbackPayload):
    client = HermesSellClient.from_env()
    # Encriptar token antes de guardar
    encrypted_token = fernet.encrypt(payload.access_token.encode()).decode()
    tenant = client.create_tenant(
        name=payload.business_name,
        phone_number_id=payload.phone_number_id,
        access_token=encrypted_token,
        waba_id=payload.waba_id,
    )
    return {
        'tenant_id': tenant.id,
        'status': 'active',
        'dashboard_url': f'/dashboard/{tenant.id}',
    }

import subprocess, shutil, uuid
from pathlib import Path
from .agent.soul import SOULBuilder
class TenantManager:
    def create(self, name, phone_number_id, access_token, waba_id,
               soul_config, openrouter_key) -> 'Tenant':
        tenant_id = str(uuid.uuid4())[:8]
        tenant_dir = self.tenants_path / tenant_id
        # 1. Crear estructura de directorios
        (tenant_dir / '.hermes' / 'memories').mkdir(parents=True)
        (tenant_dir / '.hermes' / 'skills').mkdir(parents=True)
        # 2. Copiar skills del SaaS
        for skill in Path('skills').iterdir():
            if skill.is_dir():
                shutil.copytree(skill, tenant_dir / '.hermes' / 'skills' / skill.name)
        # 3. Generar config.yaml del tenant
        config = self._render_config(tenant_id, openrouter_key)
        (tenant_dir / '.hermes' / 'config.yaml').write_text(config)
        # 4. Generar SOUL.md
        soul = SOULBuilder(soul_config).build()
        (tenant_dir / '.hermes' / 'SOUL.md').write_text(soul)
        # 5. Generar .env del tenant
        env_content = self._render_env(tenant_id, phone_number_id, access_token, waba_id)
        (tenant_dir / '.env').write_text(env_content)
        # 6. Levantar contenedor Docker
        subprocess.run([
            'docker', 'run', '-d',
            '--name', f'hermes-{tenant_id}',
            '--env-file', str(tenant_dir / '.env'),
            '-v', f'{tenant_dir}/.hermes:/home/hermes/.hermes',
            '--restart', 'unless-stopped',
            'hermesell/agent:v0.13.13',
        ], check=True)
        # 7. Registrar en DB
        tenant = Tenant(id=tenant_id, name=name, ...)
        self._db.save(tenant)
        return tenant
